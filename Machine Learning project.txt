import streamlit as st
import PyPDF2
import docx
import re

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Page configuration
st.set_page_config(
    page_title="AI Resume Screener",
    page_icon="📄",
    layout="wide"
)

# Skills database
SKILLS = [
    "python", "java", "c++", "sql", "machine learning",
    "deep learning", "data science", "html", "css",
    "javascript", "react", "django", "flask",
    "communication", "teamwork", "leadership",
    "problem solving", "excel", "power bi",
    "data analysis", "cloud computing"
]

# Extract text from PDF
def extract_pdf(file):
    reader = PyPDF2.PdfReader(file)
    text = ""

    for page in reader.pages:
        text += page.extract_text() or ""

    return text


# Extract text from DOCX
def extract_docx(file):
    document = docx.Document(file)
    text = ""

    for paragraph in document.paragraphs:
        text += paragraph.text + " "

    return text


# Identify skills
def find_skills(text):
    text = text.lower()

    found = []

    for skill in SKILLS:
        pattern = r'(?<!\w)' + re.escape(skill) + r'(?!\w)'

        if re.search(pattern, text):
            found.append(skill)

    return found


# Calculate similarity
def calculate_score(resume, job):
    vectorizer = TfidfVectorizer(
        stop_words="english",
        ngram_range=(1, 2)
    )

    vectors = vectorizer.fit_transform([resume, job])

    score = cosine_similarity(
        vectors[0:1],
        vectors[1:2]
    )[0][0]

    return round(score * 100, 2)


# Main interface
st.title("🤖 AI-Based Resume Screener")
st.subheader("Smart Resume Analysis and Job Matching System")

st.write(
    "Upload your resume and enter a job description "
    "to discover how well your profile matches the job."
)

# Upload resume
uploaded_file = st.file_uploader(
    "Upload Resume (PDF or DOCX)",
    type=["pdf", "docx"]
)

# Job description
job_description = st.text_area(
    "Enter Job Description",
    height=200,
    placeholder="Enter required qualifications, skills and experience..."
)

# Analyse button
if st.button("Analyze Resume", type="primary"):

    if uploaded_file is None:
        st.error("Please upload a resume.")

    elif not job_description.strip():
        st.error("Please enter a job description.")

    else:

        # Extract resume text
        if uploaded_file.name.lower().endswith(".pdf"):
            resume_text = extract_pdf(uploaded_file)
        else:
            resume_text = extract_docx(uploaded_file)

        if not resume_text.strip():
            st.error(
                "No readable text found. "
                "Please upload a text-based resume."
            )

        else:
            # Find skills
            resume_skills = find_skills(resume_text)
            job_skills = find_skills(job_description)

            # Find matching and missing skills
            matching_skills = [
                skill for skill in job_skills
                if skill in resume_skills
            ]

            missing_skills = [
                skill for skill in job_skills
                if skill not in resume_skills
            ]

            # Calculate AI matching score
            score = calculate_score(
                resume_text,
                job_description
            )

            # Display results
            st.success("Resume analysis completed!")

            st.header("📊 Resume Matching Result")

            st.metric(
                "Job Matching Score",
                f"{score}%"
            )

            st.progress(score / 100)

            if score >= 70:
                st.success("Strong text match")
            elif score >= 40:
                st.warning("Moderate text match")
            else:
                st.error("Low text match")

            st.subheader("✅ Matching Skills")

            if matching_skills:
                st.write(", ".join(matching_skills))
            else:
                st.write("No matching skills found.")

            st.subheader("❌ Missing Skills")

            if missing_skills:
                st.write(", ".join(missing_skills))
            else:
                st.write("No missing skills found.")

            st.subheader("📌 Resume Skills")

            if resume_skills:
                st.write(", ".join(resume_skills))
            else:
                st.write("No listed skills detected.")

            st.info(
                "Note: This score measures text similarity, "
                "not the candidate's actual ability or suitability. "
                "Recruiters should review resumes manually."
            )
